# Generating a TypeScript Client from a Stellar Contract Schema

Hand-writing a TypeScript client for every Soroban contract method invites drift: the contract adds a parameter, the client doesn't, and the mismatch surfaces at runtime instead of compile time. This tutorial builds a small code generator that reads a contract's method schema and emits a typed client, so the client is regenerated — not hand-edited — whenever the contract's interface changes.

<Note type="warning">
  This tutorial generates a client from a hand-authored JSON schema describing
  a contract's methods, not from a live on-chain WASM binary. For generating
  directly from a deployed contract's on-chain interface, see the official
  `@stellar/stellar-sdk` contract client bindings generator, which this
  tutorial's approach complements for cases where you want a lighter-weight,
  version-controlled schema as the source of truth.
</Note>

## Generator setup

The generator takes a small JSON schema (one entry per contract method) and produces a single TypeScript file. Three pieces: a schema type, a type-mapping function (contract types → TypeScript types), and the template that stitches a method schema into a function.

```ts
// scripts/codegen/schema-types.ts
export type ContractFieldType =
  | "Address"
  | "String"
  | "U32"
  | "U64"
  | "I128"
  | "Bool";

export interface ContractParam {
  name: string;
  type: ContractFieldType;
}

export interface ContractMethodSchema {
  name: string;
  params: ContractParam[];
  returns: ContractFieldType | "void";
  docs?: string;
}

export interface ContractSchema {
  contractName: string;
  methods: ContractMethodSchema[];
}
```

```ts
// scripts/codegen/type-mapping.ts
import type { ContractFieldType } from "./schema-types";

const TYPE_MAP: Record<ContractFieldType, string> = {
  Address: "string",
  String: "string",
  U32: "number",
  U64: "bigint",
  I128: "bigint",
  Bool: "boolean",
};

export function mapContractType(type: ContractFieldType): string {
  return TYPE_MAP[type];
}

export function mapReturnType(type: ContractFieldType | "void"): string {
  return type === "void" ? "void" : mapContractType(type);
}
```

```ts
// scripts/codegen/type-mapping.test.ts
import { describe, expect, it } from "vitest";
import { mapContractType, mapReturnType } from "./type-mapping";

describe("mapContractType", () => {
  it("maps Address and String to string", () => {
    expect(mapContractType("Address")).toBe("string");
    expect(mapContractType("String")).toBe("string");
  });

  it("maps U32 to number", () => {
    expect(mapContractType("U32")).toBe("number");
  });

  it("maps U64 and I128 to bigint, since they can exceed Number.MAX_SAFE_INTEGER", () => {
    expect(mapContractType("U64")).toBe("bigint");
    expect(mapContractType("I128")).toBe("bigint");
  });

  it("maps Bool to boolean", () => {
    expect(mapContractType("Bool")).toBe("boolean");
  });
});

describe("mapReturnType", () => {
  it("maps void to void", () => {
    expect(mapReturnType("void")).toBe("void");
  });

  it("delegates non-void types to mapContractType", () => {
    expect(mapReturnType("U32")).toBe("number");
  });
});
```

## The generator

```ts
// scripts/codegen/generate-client.ts
import type { ContractMethodSchema, ContractSchema } from "./schema-types";
import { mapContractType, mapReturnType } from "./type-mapping";

function generateMethodSignature(method: ContractMethodSchema): string {
  const params = method.params
    .map((p) => `${p.name}: ${mapContractType(p.type)}`)
    .join(", ");
  return `${method.name}(${params}): Promise<${mapReturnType(method.returns)}>`;
}

function generateMethodBody(method: ContractMethodSchema): string {
  const argNames = method.params.map((p) => p.name).join(", ");
  return [
    `  async ${generateMethodSignature(method)} {`,
    method.docs ? `    // ${method.docs}` : null,
    `    return this.invoke("${method.name}", [${argNames}]);`,
    `  }`,
  ]
    .filter((line): line is string => line !== null)
    .join("\n");
}

export function generateClient(schema: ContractSchema): string {
  const methodBodies = schema.methods.map(generateMethodBody).join("\n\n");

  return `// AUTO-GENERATED — do not edit by hand. Regenerate from the contract schema.
import { Contract, TransactionBuilder } from "@stellar/stellar-sdk";

export class ${schema.contractName}Client {
  constructor(private readonly contract: Contract) {}

  private async invoke(method: string, args: unknown[]): Promise<unknown> {
    // Wraps contract.call + TransactionBuilder submission; abbreviated here.
    throw new Error("invoke() must be implemented by the runtime client base");
  }

${methodBodies}
}
`;
}
```

```ts
// scripts/codegen/generate-client.test.ts
import { describe, expect, it } from "vitest";
import { generateClient } from "./generate-client";
import type { ContractSchema } from "./schema-types";

const insuranceSchema: ContractSchema = {
  contractName: "InsurancePool",
  methods: [
    {
      name: "buy_policy",
      params: [
        { name: "holder", type: "Address" },
        { name: "coverage", type: "I128" },
      ],
      returns: "U32",
      docs: "Purchases a new policy and returns its policy id.",
    },
    {
      name: "get_policy",
      params: [{ name: "policyId", type: "U32" }],
      returns: "String",
    },
  ],
};

describe("generateClient", () => {
  it("names the generated class after the contract", () => {
    const output = generateClient(insuranceSchema);
    expect(output).toContain("class InsurancePoolClient");
  });

  it("generates a correctly-typed method signature for each schema method", () => {
    const output = generateClient(insuranceSchema);
    expect(output).toContain(
      "async buy_policy(holder: string, coverage: bigint): Promise<number>",
    );
    expect(output).toContain(
      "async get_policy(policyId: number): Promise<string>",
    );
  });

  it("includes the docs comment when provided", () => {
    const output = generateClient(insuranceSchema);
    expect(output).toContain("// Purchases a new policy and returns its policy id.");
  });

  it("omits a docs comment line when none is provided", () => {
    const output = generateClient(insuranceSchema);
    const getPolicySection = output.split("async get_policy")[1];
    expect(getPolicySection.split("\n")[1].trim()).not.toMatch(/^\/\//);
  });

  it("marks the output as auto-generated", () => {
    const output = generateClient(insuranceSchema);
    expect(output).toContain("AUTO-GENERATED — do not edit by hand");
  });
});
```

## Output structure

The generator writes one file per contract, alongside its schema, so the two stay next to each other in version control:

```text
contracts/
  insurance-pool/
    schema.json              ← hand-authored, source of truth
    generated/
      InsurancePoolClient.ts ← generated, never hand-edited
```

A small CLI wrapper ties schema-in to file-out and is what CI actually runs:

```ts
// scripts/codegen/cli.ts
import { readFileSync, writeFileSync } from "fs";
import { generateClient } from "./generate-client";
import type { ContractSchema } from "./schema-types";

const [schemaPath, outputPath] = process.argv.slice(2);
if (!schemaPath || !outputPath) {
  console.error("Usage: tsx cli.ts <schema.json> <output.ts>");
  process.exit(1);
}

const schema: ContractSchema = JSON.parse(readFileSync(schemaPath, "utf-8"));
writeFileSync(outputPath, generateClient(schema));
console.log(`Generated ${outputPath} from ${schemaPath}`);
```

```bash
npx tsx scripts/codegen/cli.ts contracts/insurance-pool/schema.json contracts/insurance-pool/generated/InsurancePoolClient.ts
```

**Commit the generated output, don't gitignore it.** A generated client that isn't checked in means every contributor needs the generator toolchain just to build the project, and a CI diff check (`git diff --exit-code` after regenerating) is the cheapest way to catch a schema edit that someone forgot to regenerate from.
